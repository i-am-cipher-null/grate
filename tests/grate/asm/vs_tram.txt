/*
 * Vertex shader that exports a known constant vector as the varying.
 *
 * export register 7 is the second enabled export, and the linker addresses it
 * as export1 (the enabled-export slot, not the register number). The position
 * is passed through unchanged. Every vertex exports the same vector, so the
 * interpolated varying is that vector at any sample point and the TRAM readback
 * cannot depend on the pixel.
 */

.attributes
	[0] = "position";
	[1] = "color";

.exports
	[0] = "gl_Position";
	[7] = "vcolor";

.constants
	[100].x = 0.25;
	[100].y = 0.5;
	[100].z = 0.75;
	[100].w = 0.875;

.asm

EXEC(export[0]=vector)
	MOVv r63.xyzw, a[0].xyzw
	NOPs
;

EXEC(export[7]=vector)
	MOVv r63.xyzw, c[100].xyzw
	NOPs
;
